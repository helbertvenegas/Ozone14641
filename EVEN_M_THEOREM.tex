% Conditional theorem to add after Propositions MNP and OWZ.
% Depends explicitly on the correctness of the source manuscript's
% monomiality, crossed-product restriction criterion and tab:split.
\begin{proposition}\label{prop:two-param-even-m-structure}
Let $B$ be a PI member of one of the six families
$\mathbb M,\mathbb N,\mathbb O,\mathbb P,\mathbb W,\mathbb Z$.
With the notation of Theorem~A, put $m=\operatorname{ord}(u^2)$ and
$m'=\operatorname{ord}(v^2)$. If $m$ is even, then
\[
 \operatorname{Oz}(B)\cong
 C_2\times C_{2m'}\times C_{2m'}.
\]
In particular the isomorphism type is independent of $m$ in this stratum.
\end{proposition}
\begin{proof}
When $m$ is even, Propositions~\ref{prop:MNP} and~\ref{prop:OWZ}
give $|\widehat G_0|=1$, since $\operatorname{ord}(u)$ is divisible by
four. Theorem~\ref{thm:criterion} therefore identifies
$\operatorname{Oz}(B)$ with the character subgroup $N_K$.
By Table~\ref{tab:split}, the grading group $\Gamma$ is generated by
$\alpha,\beta,\delta$, where $2\delta=0$, and $\Gamma_K$ is generated
by $m\delta,2m'\alpha,2m'\beta$. Since $m$ is even, $m\delta=0$ is
redundant. Thus
\[
 \Gamma/\Gamma_K\cong C_{2m'}\times C_{2m'}\times C_2.
\]
The character group of a finite abelian group over an algebraically
closed field of characteristic zero has the same invariant factors.
Hence $\operatorname{Oz}(B)\cong C_{2m'}^2\times C_2$.
\end{proof}
